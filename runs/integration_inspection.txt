class CascadePressureController:
    """PI pressure controller that adjusts boiler fuel demand.

    Pressure is measured in per-unit (pu). The controller output is a
    fuel command in pu, centered around the nominal fuel demand.

    This is an educational simulation controller, not validated
    industrial combustion-control logic.
    """

    def __init__(
        self,
        setpoint_pu=1.0,
        nominal_fuel_command_pu=1.0,
        pressure_kp=3.0,
        pressure_ki=0.15,
    ):
        self.setpoint_pu = float(setpoint_pu)
        self.nominal_fuel_command_pu = float(nominal_fuel_command_pu)

        if not 0.0 <= self.nominal_fuel_command_pu <= 1.5:
            raise ValueError(
                "nominal_fuel_command_pu must be between 0.0 and 1.5"
            )

        self.pressure_pid = PIDController(
            kp=pressure_kp,
            ki=pressure_ki,
            kd=0.0,
            output_min=-1.0,
            output_max=0.5,
        )

    def reset(self):
        """Reset pressure-controller history."""
        self.pressure_pid.reset()

    def update(self, measured_pressure_pu, dt):
        """Return a bounded fuel-demand command in per-unit."""
        correction_pu = self.pressure_pid.update(
            setpoint=self.setpoint_pu,
            measurement=measured_pressure_pu,
            dt=dt,
        )

        fuel_command_pu = (
            self.nominal_fuel_command_pu + correction_pu
        )

        return max(0.0, min(1.5, fuel_command_pu))

"""PID controller with output saturation and conditional anti-windup."""

from dataclasses import dataclass


@dataclass
class PIDController:
    """Discrete-time PID controller.

    The controller uses error = setpoint - measurement.
    Output is constrained to [output_min, output_max].
    Conditional integration limits integral windup.
    """

    kp: float
    ki: float
    kd: float = 0.0
    output_min: float = 0.0
    output_max: float = 1.5
    derivative_tau: float = 0.0

    def __post_init__(self):
        if self.output_min >= self.output_max:
            raise ValueError("output_min must be less than output_max")
        if self.derivative_tau < 0:
            raise ValueError("derivative_tau cannot be negative")

        self.reset()

    def reset(self, output: float = 0.0) -> None:
        """Reset controller history."""
        self.integral = 0.0
        self.previous_error = None
        self.filtered_derivative = 0.0
        self.last_output = self._clamp(output)

    def _clamp(self, value: float) -> float:
        return max(self.output_min, min(self.output_max, value))

    def update(self, setpoint: float, measurement: float, dt: float) -> float:
        """Calculate a bounded controller output for one time step."""
        if dt <= 0:
            raise ValueError("dt must be greater than zero")

        error = float(setpoint) - float(measurement)

        if self.previous_error is None:
            raw_derivative = 0.0
        else:
            raw_derivative = (error - self.previous_error) / dt

        # Optional first-order filtering of the derivative term.
        if self.derivative_tau > 0:
            alpha = dt / (self.derivative_tau + dt)
            self.filtered_derivative += alpha * (
                raw_derivative - self.filtered_derivative
            )
        else:
            self.filtered_derivative = raw_derivative

        proportional = self.kp * error
        derivative = self.kd * self.filtered_derivative

        # Trial output before updating the integral term.
        trial_integral = self.integral + error * dt
        trial_unbounded = (
            proportional + self.ki * trial_integral + derivative
        )
        trial_output = self._clamp(trial_unbounded)

        # Integrate when unsaturated, or when the error pushes the
        # saturated output back toward the permitted range.
        pushing_further_high = (
            trial_unbounded > self.output_max and error * self.ki > 0
        )
        pushing_further_low = (
            trial_unbounded < self.output_min and error * self.ki < 0
        )

        if not (pushing_further_high or pushing_further_low):
            self.integral = trial_integral

        unbounded = (
            proportional + self.ki * self.integral + derivative
        )
        output = self._clamp(unbounded)

        self.previous_error = error
        self.last_output = output
        return output

from control.pid import PIDController


class SingleElementController:
    """Use drum-level feedback to regulate feedwater flow.

    The PI controller generates a correction around the nominal
    feedwater command of 1.0 pu.

    Inputs:
        setpoint_pct: Desired drum level (%)
        measured_level_pct: Measured drum level (%)
        dt: Simulation time step (s)

    Output:
        Feedwater command in per-unit, limited to 0.0â€“1.5 pu.
    """

    def __init__(
        self,
        setpoint_pct=50.0,
        kp=0.08,
        ki=0.003,
        nominal_feedwater_pu=1.0,
    ):
        self.setpoint_pct = float(setpoint_pct)
        self.nominal_feedwater_pu = float(nominal_feedwater_pu)

        # PID output represents a correction to nominal feedwater.
        self.pid = PIDController(
            kp=kp,
            ki=ki,
            kd=0.0,
            output_min=-self.nominal_feedwater_pu,
            output_max=1.5 - self.nominal_feedwater_pu,
        )

    def reset(self):
        """Reset controller history for a fresh simulation."""
        self.pid.reset()

    def update(self, measured_level_pct, dt):
        """Calculate the feedwater command from measured drum level."""
        correction = self.pid.update(
            setpoint=self.setpoint_pct,
            measurement=measured_level_pct,
            dt=dt,
        )

        command = self.nominal_feedwater_pu + correction
        return max(0.0, min(1.5, command))

"""Three-element boiler drum-level controller.

Elements:
1. Drum-level feedback: outer PI controller.
2. Steam-flow feedforward: anticipates changes in steam demand.
3. Feedwater-flow feedback: inner PI controller tracks required flow.
"""

from control.pid import PIDController


class ThreeElementController:
    """Three-element drum-level controller.

    All flows use per-unit (pu) values. Drum level uses percent.
    The returned value is the feedwater valve command in pu.
    """

    def __init__(
        self,
        setpoint_pct=50.0,
        level_kp=0.025,
        level_ki=0.001,
        flow_kp=1.5,
        flow_ki=0.8,
    ):
        self.setpoint_pct = float(setpoint_pct)

        # Outer loop: level error produces a flow trim.
        self.level_pid = PIDController(
            kp=level_kp,
            ki=level_ki,
            kd=0.0,
            output_min=-0.5,
            output_max=0.5,
        )

        # Inner loop: feedwater flow error adjusts the valve command.
        self.flow_pid = PIDController(
            kp=flow_kp,
            ki=flow_ki,
            kd=0.0,
            output_min=-1.5,
            output_max=1.5,
        )

    def reset(self):
        """Reset both controller loops."""
        self.level_pid.reset()
        self.flow_pid.reset()

    def update(
        self,
        measured_level_pct,
        measured_steam_flow_pu,
        measured_feedwater_flow_pu,
        dt,
    ):
        """Calculate the feedwater valve command."""

        # Element 1: level controller produces a trim in pu.
        level_trim_pu = self.level_pid.update(
            setpoint=self.setpoint_pct,
            measurement=measured_level_pct,
            dt=dt,
        )

        # Element 2: steam-flow feedforward plus level trim.
        desired_feedwater_flow_pu = (
            float(measured_steam_flow_pu) + level_trim_pu
        )

        # The desired flow must stay within the plant's flow range.
        desired_feedwater_flow_pu = max(
            0.0, min(1.5, desired_feedwater_flow_pu)
        )

        # Element 3: inner flow loop tracks the desired feedwater flow.
        flow_correction = self.flow_pid.update(
            setpoint=desired_feedwater_flow_pu,
            measurement=measured_feedwater_flow_pu,
            dt=dt,
        )

        # Use measured flow as the nominal valve-command estimate.
        command = float(measured_feedwater_flow_pu) + flow_correction

        return max(0.0, min(1.5, command))

    }

    return build_register_image(values)

"""Scaled 16-bit Modbus holding-register map for the boiler digital twin."""

from math import isfinite


REGISTER_MAP = {
    "drum_level_pct": {"address": 0, "scale": 100.0},
    "pressure_pu": {"address": 1, "scale": 10000.0},
    "steam_flow_pu": {"address": 2, "scale": 1000.0},
    "feedwater_flow_pu": {"address": 3, "scale": 1000.0},
    "fuel_flow_pu": {"address": 4, "scale": 1000.0},
    "heat_duty_pu": {"address": 5, "scale": 1000.0},
    "feedwater_command_pu": {"address": 6, "scale": 1000.0},
    "fuel_command_pu": {"address": 7, "scale": 1000.0},
    "trip_latched": {"address": 8, "scale": 1.0},
    "fuel_permissive": {"address": 9, "scale": 1.0},
    "alarm_active": {"address": 10, "scale": 1.0},
    "trip_cause_mask": {"address": 11, "scale": 1.0},
    "simulation_status": {"address": 12, "scale": 1.0},
}

REGISTER_COUNT = max(
    item["address"] for item in REGISTER_MAP.values()
) + 1

TRIP_CAUSE_BITS = {
    "LOW_LOW_DRUM_LEVEL": 0,
    "HIGH_HIGH_DRUM_LEVEL": 1,
    "LOW_LOW_PRESSURE": 2,
    "HIGH_HIGH_PRESSURE": 3,
    "LEVEL_SIGNAL_OUT_OF_RANGE": 4,
    "PRESSURE_SIGNAL_OUT_OF_RANGE": 5,
    "SENSOR_INVALID": 6,
    "COMMUNICATION_LOSS": 7,
}


def encode_value(name, value):
    """Convert one named value to an unsigned 16-bit register value."""
    if name not in REGISTER_MAP:
        raise KeyError(f"Unknown register name: {name}")

    if isinstance(value, bool):
        numeric_value = int(value)
    else:
        try:
            numeric_value = float(value)
        except (TypeError, ValueError, OverflowError) as exc:
            raise ValueError(f"Invalid value for {name}") from exc

    if not isfinite(numeric_value):
        raise ValueError(f"Non-finite value for {name}")

    scaled = round(numeric_value * REGISTER_MAP[name]["scale"])

    if not 0 <= scaled <= 65535:
        raise ValueError(
            f"Encoded value for {name} is outside unsigned 16-bit range: "
            f"{scaled}"
        )

    return int(scaled)


def decode_value(name, register_value):
    """Convert one unsigned 16-bit register value to engineering units."""
    if name not in REGISTER_MAP:
        raise KeyError(f"Unknown register name: {name}")

    if isinstance(register_value, bool) or not isinstance(register_value, int):
        raise ValueError("Register value must be an integer")

    if not 0 <= register_value <= 65535:
        raise ValueError("Register value must be between 0 and 65535")

    return register_value / REGISTER_MAP[name]["scale"]


def encode_trip_causes(causes):
    """Encode recognized trip causes into a 16-bit bitmask."""
    mask = 0
    for cause in causes:
        if cause not in TRIP_CAUSE_BITS:
            raise ValueError(f"Unknown trip cause: {cause}")
        mask |= 1 << TRIP_CAUSE_BITS[cause]
    return mask


def build_register_image(values):
    """Build a complete address-indexed holding-register image.

    Every defined register must be provided. Unknown names are rejected.
    """
    unknown = set(values) - set(REGISTER_MAP)
    missing = set(REGISTER_MAP) - set(values)

    if unknown:
        raise KeyError(f"Unknown register names: {sorted(unknown)}")
    if missing:
        raise KeyError(f"Missing register names: {sorted(missing)}")

    registers = [0] * REGISTER_COUNT

    for name, value in values.items():
        registers[REGISTER_MAP[name]["address"]] = encode_value(name, value)

    return registers

"""Educational boiler safety interlocks; not a certified safety system."""

from dataclasses import dataclass
from math import isfinite


@dataclass(frozen=True)
class SafetyConfig:
    low_low_level_pct: float = 20.0
    high_high_level_pct: float = 80.0
    low_low_pressure_pu: float = 0.5
    high_high_pressure_pu: float = 1.5


@dataclass(frozen=True)
class SafetyStatus:
    trip_latched: bool
    fuel_permissive: bool
    fuel_command_pu: float
    alarm_active: bool
    trip_causes: tuple[str, ...]
    reset_accepted: bool


class SafetyInterlock:
    """Latched protective logic for an educational boiler simulation."""

    def __init__(self, config=None):
        self.config = config if config is not None else SafetyConfig()
        self._validate_config()
        self.trip_latched = False
        self.trip_causes = ()

    def _validate_config(self):
        c = self.config
        values = (
            c.low_low_level_pct,
            c.high_high_level_pct,
            c.low_low_pressure_pu,
            c.high_high_pressure_pu,
        )

        if not all(isfinite(value) for value in values):
            raise ValueError("Safety thresholds must be finite numbers")

        if not (
            0.0 <= c.low_low_level_pct < c.high_high_level_pct <= 100.0
        ):
            raise ValueError("Invalid drum-level trip thresholds")

        if not (
            0.0 <= c.low_low_pressure_pu < c.high_high_pressure_pu
        ):
            raise ValueError("Invalid pressure trip thresholds")

    def evaluate(
        self,
        level_pct,
        pressure_pu,
        *,
        sensors_valid=True,
        communication_ok=True,
        manual_reset=False,
    ):
        causes = []

        if not sensors_valid:
            causes.append("SENSOR_INVALID")

        if not communication_ok:
            causes.append("COMMUNICATION_LOSS")

        if sensors_valid:
            try:
                level = float(level_pct)
                pressure = float(pressure_pu)
            except (TypeError, ValueError, OverflowError):
                causes.append("SENSOR_INVALID")
            else:
                if not isfinite(level) or not isfinite(pressure):
                    causes.append("SENSOR_INVALID")
                else:
                    if not 0.0 <= level <= 100.0:
                        causes.append("LEVEL_SIGNAL_OUT_OF_RANGE")

                    if pressure < 0.0:
                        causes.append("PRESSURE_SIGNAL_OUT_OF_RANGE")

                    if level <= self.config.low_low_level_pct:
                        causes.append("LOW_LOW_DRUM_LEVEL")

                    if level >= self.config.high_high_level_pct:
                        causes.append("HIGH_HIGH_DRUM_LEVEL")

                    if pressure <= self.config.low_low_pressure_pu:
                        causes.append("LOW_LOW_PRESSURE")

                    if pressure >= self.config.high_high_pressure_pu:
                        causes.append("HIGH_HIGH_PRESSURE")

        causes = tuple(dict.fromkeys(causes))

        if causes:
            self.trip_latched = True
            self.trip_causes = tuple(
                dict.fromkeys(self.trip_causes + causes)
            )

        reset_accepted = False

        if self.trip_latched and manual_reset and not causes:
            self.trip_latched = False
            self.trip_causes = ()
            reset_accepted = True

        fuel_permissive = not self.trip_latched

        return SafetyStatus(
            trip_latched=self.trip_latched,
            fuel_permissive=fuel_permissive,
            fuel_command_pu=1.0 if fuel_permissive else 0.0,
            alarm_active=self.trip_latched,
            trip_causes=self.trip_causes,
            reset_accepted=reset_accepted,
        )

"""Compare single-element and three-element boiler level control."""

import csv
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

from model.boiler import BoilerModel
from control import SingleElementController, ThreeElementController

ROOT = Path(__file__).resolve().parent
RUNS = ROOT / "runs"
PLOTS = RUNS / "plots"
PLOTS.mkdir(parents=True, exist_ok=True)

DURATION_S = 60.0
DISTURBANCE_TIME_S = 10.0
LEVEL_SETPOINT_PCT = 50.0


def run_scenario(controller_type):
    """Run one controller with the same initial conditions and disturbance."""
    boiler = BoilerModel()
    dt = boiler.dt

    if controller_type == "single":
        controller = SingleElementController(
            setpoint_pct=LEVEL_SETPOINT_PCT
        )
    elif controller_type == "three":
        controller = ThreeElementController(
            setpoint_pct=LEVEL_SETPOINT_PCT
        )
    else:
        raise ValueError(f"Unknown controller: {controller_type}")

    records = []
    for _ in range(round(DURATION_S / dt)):
        state = boiler.state
        demand = (
            1.0 if state.time_s < DISTURBANCE_TIME_S else 1.2
        )

        if controller_type == "single":
            command = controller.update(
                measured_level_pct=state.drum_level_pct,
                dt=dt,
            )
        else:
            command = controller.update(
                measured_level_pct=state.drum_level_pct,
                measured_steam_flow_pu=state.steam_flow_pu,
                measured_feedwater_flow_pu=state.feedwater_flow_pu,
                dt=dt,
            )

        boiler.step(
            feedwater_command_pu=command,
            steam_demand_pu=demand,
            fuel_command_pu=1.0,
        )

        s = boiler.state
        records.append({
            "time_s": s.time_s,
            "steam_demand_pu": demand,
            "level_setpoint_pct": LEVEL_SETPOINT_PCT,
            "drum_level_pct": s.drum_level_pct,
            "level_error_pct": LEVEL_SETPOINT_PCT - s.drum_level_pct,
            "inventory_pu": s.inventory_pu,
            "pressure_pu": s.pressure_pu,
            "steam_flow_pu": s.steam_flow_pu,
            "feedwater_flow_pu": s.feedwater_flow_pu,
            "feedwater_command_pu": command,
            "indicated_swell_pct": s.indicated_swell_pct,
        })

    return records


def calculate_metrics(records):
    """Calculate level-control metrics over the complete run."""
    dt = records[1]["time_s"] - records[0]["time_s"]
    errors = [r["level_error_pct"] for r in records]
    post_step = [
        r for r in records if r["time_s"] >= DISTURBANCE_TIME_S
    ]

    # Settling time: first point after which all remaining errors
    # stay within +/-1 percentage point.
    settling_time = None
    for i, row in enumerate(post_step):
        if all(abs(r["level_error_pct"]) <= 1.0
               for r in post_step[i:]):
            settling_time = row["time_s"] - DISTURBANCE_TIME_S
            break

    post_errors = [r["level_error_pct"] for r in post_step]

    return {
        "IAE_pct_s": sum(abs(e) for e in errors) * dt,
        "maximum_absolute_level_error_pct": max(
            abs(e) for e in post_errors
        ),
        "positive_level_overshoot_pct": max(
            0.0, max(-e for e in post_errors)
        ),
        "final_level_pct": records[-1]["drum_level_pct"],
        "final_level_error_pct": records[-1]["level_error_pct"],
        "final_inventory_pu": records[-1]["inventory_pu"],
        "final_pressure_pu": records[-1]["pressure_pu"],
        "settling_time_after_step_s": settling_time,
    }


def write_csv(path, records):
    with path.open("w", newline="", encoding="utf-8") as file:
        writer = csv.DictWriter(file, fieldnames=records[0].keys())
        writer.writeheader()
        writer.writerows(records)


def plot_comparison(single, three, column, ylabel, title, filename):
    plt.figure(figsize=(10, 5))

    for records, label in (
        (single, "Single-element"),
        (three, "Three-element"),
    ):
        plt.plot(
            [r["time_s"] for r in records],
            [r[column] for r in records],
            label=label,
        )

    plt.axvline(
        DISTURBANCE_TIME_S,
        linestyle="--",
        label="Steam-demand step",
    )

    if column == "drum_level_pct":
        plt.axhline(
            LEVEL_SETPOINT_PCT,
            linestyle=":",
            label="Level setpoint",
        )

    plt.xlabel("Time (s)")
    plt.ylabel(ylabel)
    plt.title(title)
    plt.grid(True, alpha=0.3)
    plt.legend()
    plt.tight_layout()
    plt.savefig(PLOTS / filename, dpi=160)
    plt.close()


def main():
    print("Running single-element simulation...")
    single = run_scenario("single")

    print("Running three-element simulation...")
    three = run_scenario("three")

    write_csv(RUNS / "single_element.csv", single)
    write_csv(RUNS / "three_element.csv", three)

    metrics = {
        "single_element": calculate_metrics(single),
        "three_element": calculate_metrics(three),
    }

    with (RUNS / "controller_comparison.csv").open(
        "w", newline="", encoding="utf-8"
    ) as file:
        writer = csv.writer(file)
        writer.writerow(["metric", "single_element", "three_element"])
        for name in metrics["single_element"]:
            writer.writerow([
                name,
                metrics["single_element"][name],
                metrics["three_element"][name],
            ])

    plot_comparison(
        single, three, "drum_level_pct", "Drum level (%)",
        "Drum-Level Control Comparison", "controller_level.png"

    plot_comparison(
        single, three, "pressure_pu", "Pressure (pu)",

import unittest

from model.boiler import BoilerModel
from control import SingleElementController, ThreeElementController


class ClosedLoopTests(unittest.TestCase):

    def simulate(self, controller_type, duration_s=60.0, step_time_s=10.0):
        boiler = BoilerModel()
        controller = controller_type(setpoint_pct=50.0)
        dt = boiler.dt
        total_steps = round(duration_s / dt)
        disturbance_step = round(step_time_s / dt)
        samples = []

        for step_index in range(total_steps):
            state = boiler.state

            # Use the integer step index to avoid floating-point time errors.
            demand = 1.0 if step_index < disturbance_step else 1.2

            if controller_type is SingleElementController:
                command = controller.update(state.drum_level_pct, dt)
            else:
                command = controller.update(
                    state.drum_level_pct,
                    state.steam_flow_pu,
                    state.feedwater_flow_pu,
                    dt,
                )

            boiler.step(
                feedwater_command_pu=command,
                steam_demand_pu=demand,
                fuel_command_pu=1.0,
            )

            samples.append({
                "time_s": boiler.state.time_s,
                "level_pct": boiler.state.drum_level_pct,
                "inventory_pu": boiler.state.inventory_pu,
                "command_pu": command,
                "demand_pu": demand,
            })

        return samples, disturbance_step

    def test_both_controllers_keep_level_within_physical_bounds(self):
        for controller_type in (
            SingleElementController,
            ThreeElementController,
        ):
            with self.subTest(controller=controller_type.__name__):
                samples, _ = self.simulate(controller_type)
                for sample in samples:
                    self.assertGreaterEqual(sample["level_pct"], 0.0)
                    self.assertLessEqual(sample["level_pct"], 100.0)

    def test_both_controllers_keep_feedwater_command_bounded(self):
        for controller_type in (
            SingleElementController,
            ThreeElementController,
        ):
            with self.subTest(controller=controller_type.__name__):
                samples, _ = self.simulate(controller_type)
                for sample in samples:
                    self.assertGreaterEqual(sample["command_pu"], 0.0)
                    self.assertLessEqual(sample["command_pu"], 1.5)

    def test_steam_demand_increases_at_ten_seconds(self):
        for controller_type in (
            SingleElementController,
            ThreeElementController,
        ):
            with self.subTest(controller=controller_type.__name__):
                samples, disturbance_step = self.simulate(controller_type)
                before = samples[:disturbance_step]
                after = samples[disturbance_step:]

                self.assertTrue(before)
                self.assertTrue(after)
                self.assertTrue(
                    all(s["demand_pu"] == 1.0 for s in before)
                )
                self.assertTrue(
                    all(s["demand_pu"] == 1.2 for s in after)
                )

    def test_three_element_final_level_is_near_setpoint(self):
        samples, _ = self.simulate(ThreeElementController)
        final_level = samples[-1]["level_pct"]
        self.assertLessEqual(abs(final_level - 50.0), 1.0)

import math
import unittest

from model.boiler import BoilerModel
from control import SingleElementController, ThreeElementController


class DisturbanceMetricTests(unittest.TestCase):
    DURATION_S = 60.0
    DISTURBANCE_STEP = 100
    SETPOINT = 50.0
    TOLERANCE_PCT = 1.0

    def simulate(self, controller_type):
        boiler = BoilerModel()
        controller = controller_type(setpoint_pct=self.SETPOINT)
        dt = boiler.dt
        total_steps = round(self.DURATION_S / dt)
        rows = []

        for step_index in range(total_steps):
            state = boiler.state
            demand = 1.0 if step_index < self.DISTURBANCE_STEP else 1.2

            if controller_type is SingleElementController:
                command = controller.update(state.drum_level_pct, dt)
            else:
                command = controller.update(
                    state.drum_level_pct,
                    state.steam_flow_pu,
                    state.feedwater_flow_pu,
                    dt,
                )

            boiler.step(
                feedwater_command_pu=command,
                steam_demand_pu=demand,
                fuel_command_pu=1.0,
            )

            s = boiler.state
            rows.append({
                "time_s": s.time_s,
                "level_pct": s.drum_level_pct,
                "error_pct": self.SETPOINT - s.drum_level_pct,
                "command_pu": command,
                "demand_pu": demand,
            })

        post = rows[self.DISTURBANCE_STEP:]
        errors = [r["error_pct"] for r in post]

        settling_time = None
        for i, row in enumerate(post):
            remaining_errors = [
                abs(r["error_pct"]) for r in post[i:]
            ]
            if all(error <= self.TOLERANCE_PCT for error in remaining_errors):
                settling_time = row["time_s"] - 10.0
                break

        return {
            "iae_pct_s": sum(abs(e) for e in errors) * dt,
            "max_abs_error_pct": max(abs(e) for e in errors),
            "final_level_error_pct": errors[-1],
            "max_command_pu": max(r["command_pu"] for r in post),
            "min_command_pu": min(r["command_pu"] for r in post),
            "settling_time_s": settling_time,
        }

    def test_all_metrics_are_finite_except_unsettled_time(self):
        for controller_type in (
            SingleElementController,
            ThreeElementController,
        ):
            with self.subTest(controller=controller_type.__name__):
                result = self.simulate(controller_type)
                for key, value in result.items():
                    if key == "settling_time_s" and value is None:
                        continue
                    self.assertTrue(
                        math.isfinite(value),
                        f"{key} is not finite: {value}",
                    )

    def test_commands_remain_within_limits_after_disturbance(self):
        for controller_type in (
            SingleElementController,
            ThreeElementController,
        ):
            with self.subTest(controller=controller_type.__name__):
                result = self.simulate(controller_type)
                self.assertGreaterEqual(result["min_command_pu"], 0.0)
                self.assertLessEqual(result["max_command_pu"], 1.5)

    def test_three_element_final_level_error_is_within_one_percent(self):
        result = self.simulate(ThreeElementController)
        self.assertLessEqual(
            abs(result["final_level_error_pct"]),
            self.TOLERANCE_PCT,
        )

    def test_print_comparison_metrics(self):
        for controller_type in (
            SingleElementController,
            ThreeElementController,
        ):
            result = self.simulate(controller_type)
            print(f"\n{controller_type.__name__}")
            for key, value in result.items():
                print(f"  {key}: {value}")
